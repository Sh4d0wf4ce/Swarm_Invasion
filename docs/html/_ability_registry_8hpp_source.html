82 static void loadConfig(const std::string& filepath) {
83 std::ifstream file(filepath);
84 if (!file.is_open()) {
85 std::cerr << "[REGISTRY ERROR] Could not open " << filepath << "!\n";
86 return;
87 }
88
89 nlohmann::json json;
90 try {
91 file >> json;
92 } catch (const std::exception& e) {
93 std::cerr << "[REGISTRY ERROR] Failed to parse " << filepath << ": " << e.what() << "\n";
94 return;
95 }
96 file.close();
97 m_stats.clear();
98 m_vanguard = {};
99 m_medic = {};
100 m_juggernaut = {};
101 m_soldier = {};
102 int abilityCount = 0;
103 for (auto& [classKey, classVal] : json.items()) {
104 if (!classVal.is_object()) continue;
105 if (classKey == "Vanguard") {
106 abilityCount += loadClass(classVal, m_vanguard);
107 } else if (classKey == "Medic") {
108 abilityCount += loadClass(classVal, m_medic);
109 } else if (classKey == "Juggernaut") {
110 abilityCount += loadClass(classVal, m_juggernaut);
111 } else if (classKey == "Soldier") {
112 abilityCount += loadClass(classVal, m_soldier);
113 } else {
114 std::cerr << "[REGISTRY WARNING] Unknown class key: " << classKey << "\n";
115 }
116 }
117
118 std::cout << "[REGISTRY] Successfully loaded " << filepath
119 << " (4 classes, " << abilityCount << " abilities)\n";
120 }

317 static AbilityStats parseEntry(const nlohmann::json& val) {
318 AbilityStats stats;
319 stats.cooldown = val.value("cooldown", 0.f);
320 stats.damage = val.value("damage", 0.f);
321 stats.knockback = val.value("knockback", 0.f);
322 stats.radius = val.value("radius", 0.f);
323 stats.range = val.value("range", 0.f);
324 stats.duration = val.value("duration", 0.f);
325 stats.speed = val.value("speed", 0.f);
326 stats.lifetime = val.value("lifetime", 0.f);
327 stats.heal = val.value("heal", 0.f);
328 static const std::unordered_set<std::string> knownFields = {
329 "cooldown", "damage", "knockback", "radius", "range",
330 "duration", "speed", "lifetime", "heal"
331 };
332 for (auto& [key, value] : val.items()) {
333 if (knownFields.count(key) || !value.is_number()) continue;
334 stats.customParams[key] = value.get<float>();
335 }
336 return stats;
337 }

346 static float paramFromStats(const AbilityStats& stats, const std::string& key, float defaultVal) {
347 auto it = stats.customParams.find(key);

350 if (key == "damage") return stats.damage;
351 if (key == "knockback") return stats.knockback;
352 if (key == "radius") return stats.radius;
353 if (key == "range") return stats.range;
354 if (key == "duration") return stats.duration;
355 if (key == "speed") return stats.speed;
356 if (key == "lifetime") return stats.lifetime;
357 if (key == "heal") return stats.heal;
358 return defaultVal;
