145 static std::vector<std::string> pickWeightedRandom(
146 const std::vector<const UpgradeDefinition*>& pool,
147 int count) {
148 std::vector<std::string> result;
149 if (pool.empty() || count <= 0) return result;
150 std::vector<const UpgradeDefinition*> remaining = pool;
151 int picks = std::min(count, static_cast<int>(remaining.size()));
152 for (int i = 0; i < picks; ++i) {
153 float totalWeight = 0.0f;
154 for (const auto* def : remaining) {
155 totalWeight += def->weight;
156 }
157 if (totalWeight <= 0.0f) break;
158 float roll = (static_cast<float>(std::rand()) / RAND_MAX) * totalWeight;
159 float cumulative = 0.0f;
160 auto chosenIt = remaining.end();
161 for (auto it = remaining.begin(); it != remaining.end(); ++it) {
162 cumulative += (*it)->weight;
163 if (roll <= cumulative) {
164 chosenIt = it;
165 break;
166 }
167 }
168 if (chosenIt == remaining.end()) {
169 chosenIt = remaining.begin();
170 }
171 result.push_back((*chosenIt)->id);
172 remaining.erase(chosenIt);
173 }
174 return result;
175 }
