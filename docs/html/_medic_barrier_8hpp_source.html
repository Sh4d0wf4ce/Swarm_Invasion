113 void buildArcMesh() {
114 constexpr int segments = 36;
115 const auto& barrier = AbilityRegistry::medic().Barrier;
116 const float span = AbilityRegistry::param(barrier, "span", 0.80f);
117 const float arcRadius = AbilityRegistry::param(barrier, "arcRadius", 280.f);
118 const float wallThickness = AbilityRegistry::param(barrier, "wallThickness", 18.f);
119 const float halfSpan = span / 2.0f;
120 const float innerR = arcRadius - wallThickness / 2.0f;
121 const float outerR = arcRadius + wallThickness / 2.0f;
122 const float startAngle = m_facingAngle - halfSpan;
123 const float endAngle = m_facingAngle + halfSpan;
124
125 m_fillVertices.setPrimitiveType(sf::PrimitiveType::TriangleStrip);
126 m_fillVertices.resize(static_cast<std::size_t>((segments + 1) * 2));
127
128 for (int i = 0; i <= segments; ++i) {
129 const float t = static_cast<float>(i) / segments;
130 const float angle = startAngle + t * (endAngle - startAngle);
131 const float cosA = std::cos(angle);
132 const float sinA = std::sin(angle);
133
134 sf::Vector2f inner(m_center.x + cosA * innerR, m_center.y + sinA * innerR);
135 sf::Vector2f outer(m_center.x + cosA * outerR, m_center.y + sinA * outerR);
136
137 m_fillVertices[static_cast<std::size_t>(i * 2)] = sf::Vertex(inner, sf::Color(50, 255, 200, 180));
138 m_fillVertices[static_cast<std::size_t>(i * 2 + 1)] = sf::Vertex(outer, sf::Color(100, 255, 255, 180));
139 }
140
141 m_outlineVertices.setPrimitiveType(sf::PrimitiveType::LineStrip);
142 m_outlineVertices.resize(static_cast<std::size_t>(segments + 1));
143
144 for (int i = 0; i <= segments; ++i) {
145 const float t = static_cast<float>(i) / segments;
146 const float angle = startAngle + t * (endAngle - startAngle);
147
148 sf::Vector2f outer(
149 m_center.x + std::cos(angle) * outerR,
150 m_center.y + std::sin(angle) * outerR
151 );
152
153 m_outlineVertices[static_cast<std::size_t>(i)] = sf::Vertex(outer, sf::Color(100, 255, 255, 255));
154 }
